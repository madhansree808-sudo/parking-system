Q3 · Count Slots by Status per Area

Aggregated occupancy summary grouped by parking area 
SELECT  a.area_name,
        COUNT(CASE WHEN s.status = 'AVAILABLE'    THEN 1 END) AS available,
        COUNT(CASE WHEN s.status = 'OCCUPIED'     THEN 1 END) AS occupied,
        COUNT(CASE WHEN s.status = 'MAINTENANCE'  THEN 1 END) AS maintenance,
        COUNT(*) AS total
FROM    parking_slots s
JOIN    parking_areas a ON s.area_id = a.area_id
GROUP BY a.area_name
ORDER BY a.area_name;

Q6 · Calculate Parking Fee for a Ticket

Fee = Base charge + (hourly rate × hours stayed) 
SELECT  t.ticket_id,
        v.license_plate,
        v.vehicle_type,
        t.entry_time,
        t.exit_time,
        ROUND(
          (JULIANDAY(t.exit_time) - JULIANDAY(t.entry_time)) * 24, 2
        ) AS hours_stayed,
        t.fee_amount AS total_fee_inr
FROM    parking_tickets t
JOIN    vehicles v ON t.vehicle_id = v.vehicle_id
WHERE   t.status = 'PAID'
ORDER BY t.exit_time DESC
LIMIT   6;

Q7 · Total Revenue by Vehicle Type

Aggregated earnings grouped by vehicle category 
SELECT  v.vehicle_type,
        COUNT(t.ticket_id)       AS total_trips,
        SUM(t.fee_amount)         AS total_revenue_inr,
        ROUND(AVG(t.fee_amount),2) AS avg_fee_inr
FROM    parking_tickets t
JOIN    vehicles v ON t.vehicle_id = v.vehicle_id
WHERE   t.status = 'PAID'
GROUP BY v.vehicle_type
ORDER BY total_revenue_inr DESC;

Q10 · Daily Parking Traffic Report

Number of vehicles parked per day with daily revenue 
SELECT  DATE(t.entry_time)        AS parking_date,
        COUNT(t.ticket_id)       AS vehicles_parked,
        SUM(t.fee_amount)         AS daily_revenue_inr
FROM    parking_tickets t
WHERE   t.status = 'PAID'

Q11 · DDL — CREATE TABLE Statements

Table definitions with primary keys, foreign keys, and constraints 
-- Table 1: Parking Areas
CREATE TABLE parking_areas (
    area_id      INTEGER PRIMARY KEY AUTOINCREMENT,
    area_name    TEXT    NOT NULL UNIQUE,
    location     TEXT    NOT NULL,
    total_slots  INTEGER DEFAULT 0,
    area_type    TEXT    CHECK(area_type IN ('Open','Covered','EV Charging'))
);

-- Table 2: Parking Slots
CREATE TABLE parking_slots (
    slot_id     INTEGER PRIMARY KEY AUTOINCREMENT,
    area_id     INTEGER NOT NULL REFERENCES parking_areas(area_id),
    slot_number TEXT    NOT NULL UNIQUE,
    slot_type   TEXT    CHECK(slot_type IN ('Two-Wheeler','Four-Wheeler','EV','Heavy')),
    status      TEXT    DEFAULT 'AVAILABLE'
                        CHECK(status IN ('AVAILABLE','OCCUPIED','MAINTENANCE')),
    floor_level TEXT    DEFAULT 'G'
);

-- Table 3: Vehicles
CREATE TABLE vehicles (
    vehicle_id     TEXT    PRIMARY KEY,
    license_plate  TEXT    NOT NULL UNIQUE,
    vehicle_type   TEXT    CHECK(vehicle_type IN ('Two-Wheeler','Four-Wheeler','EV','Heavy')),
    owner_name     TEXT    NOT NULL,
    contact_number TEXT,
    registered_at  DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Table 4: Parking Tickets
CREATE TABLE parking_tickets (
    ticket_id   TEXT     PRIMARY KEY,
    vehicle_id  TEXT     NOT NULL REFERENCES vehicles(vehicle_id),
    slot_id     INTEGER  NOT NULL REFERENCES parking_slots(slot_id),
    entry_time  DATETIME DEFAULT CURRENT_TIMESTAMP,
    exit_time   DATETIME,
    fee_amount  REAL     DEFAULT 0.0,
    status      TEXT     DEFAULT 'ACTIVE'
                         CHECK(status IN ('ACTIVE','PAID','CANCELLED'))
);
